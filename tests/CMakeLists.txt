cmake_minimum_required(VERSION 3.19)
project(lcars_ui_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../firmware/.esphome/build/lcars-controller/managed_components/lvgl__lvgl" lvgl)
add_executable(ui_smoke ui_smoke.cpp)
target_link_libraries(ui_smoke PRIVATE lvgl)
enable_testing()
set(LCARS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
file(MAKE_DIRECTORY "${LCARS_ROOT}/logs")

# Reuse the cJSON version from the firmware SDK; do not download a different
# parser for host tests. ESPHome produces this description when it compiles.
set(LCARS_DESCRIPTION "${LCARS_ROOT}/firmware/.esphome/build/lcars-controller/build/project_description.json")
if(NOT EXISTS "${LCARS_DESCRIPTION}")
    message(FATAL_ERROR "Compile firmware/lcars.yaml with ESPHome before configuring native tests.")
endif()
file(READ "${LCARS_DESCRIPTION}" LCARS_DESCRIPTION_JSON)
string(JSON LCARS_IDF_DIR GET "${LCARS_DESCRIPTION_JSON}" idf_path)
add_library(transit_cjson STATIC "${LCARS_IDF_DIR}/components/json/cJSON/cJSON.c")
target_include_directories(transit_cjson PUBLIC "${LCARS_IDF_DIR}/components/json/cJSON")
target_link_libraries(ui_smoke PRIVATE transit_cjson)

foreach(test_name controller_state_test pcm_playback_test sd_signature_test sd_file_test transit_test weather_test)
    add_executable(${test_name} "${test_name}.cpp")
endforeach()
target_link_libraries(transit_test PRIVATE transit_cjson)
target_link_libraries(weather_test PRIVATE transit_cjson)
foreach(test_name ui_smoke controller_state_test pcm_playback_test sd_signature_test sd_file_test transit_test weather_test)
    target_compile_options(${test_name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME ${test_name} COMMAND ${test_name})
    set_tests_properties(${test_name} PROPERTIES WORKING_DIRECTORY "${LCARS_ROOT}")
endforeach()
